========================================== # 🧠 EXPERT BRAIN (МАТЕМАТИКА) # ========================================== def calculate_bollinger_bands(prices, period=20, std_dev=2): if len(prices) < period: return None # Simple Moving Average (SMA) sma = sum(prices[-period:]) / period # Standard Deviation variance = sum([((x - sma) ** 2) for x in prices[-period:]]) / period std = math.sqrt(variance) return { "upper": sma + (std * std_dev), "lower": sma - (std * std_dev), "middle": sma, "width": (sma + (std * std_dev) - (sma - (std * std_dev))) / sma # Ширина канала } def analyze_regime(prices): """Определяет состояние рынка""" bb = calculate_bollinger_bands(prices) if not bb: return "UNKNOWN" # Если ширина канала больше 5% - это ШТОРМ if bb["width"] > 0.05: return "STORM" #
